streamlit==1.60.0

# The pyarrow<20 / numpy<2.3 / protobuf<6 caps that used to live here existed
# only to hold those libraries at versions contemporary with streamlit 1.42,

# app to builds *older* than what streamlit itself expects.

# Authlib was pinned to 1.6.5 to dodge streamlit/streamlit#13461 ("400:
# 'NoneType' object does not support item assignment" on /oauth2callback).
# That pin was moved to 1.6.12 on 2026-07-26 because 1.6.5 carries nine known
# vulnerabilities, several of which land squarely on this app's threat model:
#
#   PYSEC-2026-2118 (fix 1.6.7)  a JWT with alg:none and an empty signature
#                                passed verification — affects 1.6.5 exactly
#   PYSEC-2026-287  (fix 1.6.9)  JWK header injection forges arbitrary JWTs
#   PYSEC-2026-2117 (fix 1.6.9)  OIDC ID-token validation flaw
#   PYSEC-2026-25   (fix 1.6.11) CSRF in starlette_client (streamlit 1.60 runs
#                                on starlette)
#
# Identity here comes from st.user.email, so a forgeable ID token is an
# account-takeover path — a worse risk than the compatibility bug, which was
# reported against Azure Entra while this app uses Google.
#
# ⚠ #13461 is still OPEN upstream and the OIDC callback has NO automated
# coverage — it cannot be exercised locally. Verify by actually signing in with
# Google after deploying. To roll back: pin Authlib==1.6.5 and redeploy.
Authlib==1.6.12

# Required BY Authlib's starlette integration, which is what streamlit imports
# to build the Google OAuth client. Authlib declares httpx as an optional extra
# and 1.6.5 imported it lazily, so it was never needed before — 1.6.12's
# starlette_client/__init__ pulls httpx_client eagerly, and without this the
# app raises ModuleNotFoundError the moment anyone clicks "Sign in with
# Google". Nothing local catches that: development uses the email form, so the
# OIDC path is never imported here. It took a production outage to find.
httpx>=0.27

# pyarrow: streamlit allows <25. Held at 19.0.1 by the old cap, which carries
# PYSEC-2026-113 (use-after-free reading malicious Arrow IPC files). This app
# never reads user-supplied Arrow files, so exposure is remote — but the fix is
# free.
pyarrow>=25.0.1,<26
